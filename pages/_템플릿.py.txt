import os
import sys

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from app_kit import *  # noqa: F403,E402

# 로그인 확인 + 모바일 대응 + 로그아웃 버튼 + 제목까지 한 줄로 처리
시작("자동차 유지비", "🚗", "연비와 주행거리로 월 유지비를 계산합니다")  # noqa: F405

# ---------------------------------------------------------------- 여기서부터
c1, c2, c3 = st.columns(3)  # noqa: F405
거리 = c1.number_input("월 주행거리(km)", min_value=0, value=1200, step=100)
연비 = c2.number_input("연비(km/L)", min_value=1.0, value=12.0, step=0.5)
유가 = c3.number_input("유가(원/L)", min_value=0, value=1700, step=10)

유류비 = (거리 / 연비 * 유가) if 연비 else 0

카드_줄([  # noqa: F405
    ("월 유류비", 원(유류비)),          # noqa: F405
    ("연 유류비", 원(유류비 * 12)),     # noqa: F405
], 열수=2)
